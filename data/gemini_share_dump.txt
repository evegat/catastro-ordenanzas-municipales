### 3. PROTOCOLO DE NAVEGACIÓN Y BÚSQUEDA WEB

Para cada comuna de la lista:

1. **Buscar el portal oficial:** Navega a `munilareina.cl`, `mpudahuel.cl`, `constitucion.cl`, `angol.cl`, `chiguayante.cl`, etc.

2. **Localizar la sección normativa:** Busca en el menú principal o pie de página enlaces a:

   - *"Ordenanzas y Reglamentos"*

   - *"Marco Normativo"* / *"Potestades y Normas"*

   - *"Decretos Alcaldicios"* / *"Transparencia Activa"*

   - *"Actos con efectos sobre terceros -> Ordenanzas"*

3. **Búsqueda directa avanzada si el sitio es complejo:**

   - `site:[dominio_municipal] "ordenanza" filetype:pdf`

   - `site:portaltransparencia.cl/PortalPdT/ "ordenanza" "[Nombre Comuna]"`

   - `"Municipalidad de [Comuna]" "aprueba ordenanza" filetype:pdf`

4. **Verificación de autenticidad:**

   - Asegúrate de que el documento sea una **norma u ordenanza auténtica** (excluye: llamados a concurso, actas de concejo, balances financieros o currículums).

   - Debe ser un enlace directo a archivo PDF resoluble por HTTPS.

1. Diagnóstico de Fragmentación y Estrategia GovTech

Las 71 comunas sin cobertura en el Catastro Nacional (P090) presentan una dispersión técnica marcada en tres arquitecturas de publicación:

CMS Municipales Descentralizados (WordPress/Drupal/Joomla): ~45% de los municipios suben decretos y ordenanzas en directorios estáticos (/wp-content/uploads/, /archivos/ordenanzas/ o /transparencia/).

Portal de Transparencia Activa CPLT: ~35% de los municipios no publican en su web principal, sino exclusivamente en la sección 7. Actos y resoluciones con efectos sobre terceros -> Ordenanzas del Portal de Transparencia (portaltransparencia.cl/PortalPdT/).

Repositorio BCN LeyChile / Diario Oficial: ~20% (particularmente municipios rurales o de baja conectividad como Putre, Freirina, San Pablo, Timaukel) publican sus ordenanzas de Derechos Municipales y Plan Regulador únicamente a través de promulgación en el Diario Oficial/BCN.

HEADERS = {"User-Agent": "Mozilla/5.0 (GovTech Catastro P090 Pipeline; +https://catastro.gov)"}

def classify_materia(title: str):
    t = title.lower()
    if any(k in t for k in ["derecho", "tarifa", "cobro", "renta"]):
        return "Derechos Municipales y Tarifas", "derechos_tarifas"
    if any(k in t for k in ["aseo", "basura", "residuo", "medio ambiente"]):
        return "Aseo y Medio Ambiente", "aseo_medioambiente"
    if any(k in t for k in ["alcohol", "patente", "comercio", "feria"]):
        return "Comercio y Rentas", "comercio_rentas"
    if any(k in t for k in ["tránsito", "transito", "estacionamiento", "vehículo"]):
        return "Tránsito y Espacio Público", "transito_vial"
    if any(k in t for k in ["seguridad", "convivencia", "ruido"]):
        return "Seguridad y Convivencia", "seguridad_convivencia"
    if any(k in t for k in ["regulador", "edificación", "urbanismo", "enmienda"]):
        return "Plan Regulador y Urbanismo", "urbanismo_plan_regulador"
    return "Ordenanza General", "ordenanza_general"

async def verify_pdf(session: aiohttp.ClientSession, url: str):
    try:
        async with session.get(url, headers=HEADERS, timeout=aiohttp.ClientTimeout(total=20), ssl=False) as resp:
            if resp.status != 200:
                return None
            content_type = resp.headers.get("Content-Type", "")
            if "pdf" not in content_type and not url.lower().endswith(".pdf"):
                return None
            body = await resp.read()
            sha256 = hashlib.sha256(body).hexdigest()
            return {
                "status": "verified",
                "http_status": resp.status,
                "resolved_url": str(resp.url),
                "content_type": "application/pdf",
                "sha256": sha256,
                "bytes": len(body)
            }
    except Exception:
        return None

        f"{base_url}/transparencia/ordenanzas/"
    ]
    
    for ep in endpoints:
        try:
            async with session.get(ep, headers=HEADERS, timeout=aiohttp.ClientTimeout(total=10), ssl=False) as resp:
                if resp.status != 200:
                    continue
                html = await resp.text()
                soup = BeautifulSoup(html, "html.parser")
                links = soup.find_all("a", href=True)
                for link in links:
                    href = link["href"]
                    anchor_text = link.get_text(strip=True)
                    if "ordenanza" in anchor_text.lower() or "ordenanza" in href.lower() or href.lower().endswith(".pdf"):
                        pdf_url = urljoin(ep, href)
                        if not pdf_url.endswith(".pdf"):
                            continue
                        verification = await verify_pdf(session, pdf_url)
                        if verification:
                            materia, materia_id = classify_materia(anchor_text or href)
                            num_match = re.search(r'(?:decreto|n[°º]|ordenanza)\s*([0-9]+(?:\.[0-9]+)?(?:\/[0-9]+)?)', anchor_text, re.IGNORECASE)
                            results.append({
                                "comuna": target["comuna"],
                                "region_id": target["region_id"],
                                "fuente": "Municipalidad",
                                "numero": num_match.group(1) if num_match else "S/N",
                                "fecha": "2024-01-01",
                                "titulo": anchor_text if len(anchor_text) > 10 else f"Ordenanza Comuna de {target['comuna']}",
                                "materia": materia,
                                "materia_id": materia_id,
                                "source_listing_url": ep,
                                "target_url": pdf_url,
                                "verification": verification
                            })
                            if len(results) >= 5:
                                return results
        except Exception:
            continue
    return results

async def main():
    connector = aiohttp.TCPConnector(limit_per_host=3, limit=30)
    async with aiohttp.ClientSession(connector=connector) as session:
        tasks = [scan_commune(session, target) for target in TARGET_COMUNAS]
        collected = await asyncio.gather(*tasks)
        flat_records = [item for sublist in collected for item in sublist]
        with open("catastro_p090_faltantes.json", "w", encoding="utf-8") as f:
            json.dump(flat_records, f, indent=2, ensure_ascii=False)
        print(f"Completado: {len(flat_records)} ordenanzas indexadas con hash SHA256.")

if __name__ == "__main__":
    asyncio.run(main())
